wifi_name=
wifi_password=

# Two-letter country code, so Wi-Fi uses the right channels (US, CA, GB, DE, AU, ...)
wifi_country=US


# =====================================================================
#  Everything below is OPTIONAL. You can leave it as it is.
# =====================================================================

# Set to yes if your Wi-Fi network is hidden (does not show up in lists).
wifi_hidden=no

# Name of the box on your network. The app is then at http://<name>.local
# Change it only if you have more than one Ember box.
hostname=ember

# Your time zone, e.g. America/New_York, Europe/London
timezone=

# Turn on SSH so you can log in to the box from a computer (user: emberadmin).
# Needs a password of your own, at least 6 characters.

ssh_password=
